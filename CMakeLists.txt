cmake_minimum_required(VERSION 3.20)
project(LilithLinuxBridge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets DBus)
find_package(LayerShellQt REQUIRED)
find_package(KF6IdleTime REQUIRED)
find_package(X11 REQUIRED COMPONENTS Xext)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/build")
set(MOC_FILES)
foreach(name global_shortcuts desktop_bridge native_bridge)
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/${name}.moc"
    COMMAND Qt6::moc "${CMAKE_SOURCE_DIR}/src/host/${name}.cpp" -o "${CMAKE_BINARY_DIR}/${name}.moc"
    DEPENDS "src/host/${name}.cpp" VERBATIM)
  list(APPEND MOC_FILES "${CMAKE_BINARY_DIR}/${name}.moc")
endforeach()
add_custom_target(bridge-moc DEPENDS ${MOC_FILES})
add_executable(lilith-host src/host/main.cpp src/host/x11_neutralize.cpp src/host/kwin_bridge.cpp src/host/global_shortcuts.cpp src/host/desktop_bridge.cpp src/host/tray_bridge.cpp src/host/native_bridge.cpp)
add_dependencies(lilith-host bridge-moc)
target_include_directories(lilith-host PRIVATE "${CMAKE_BINARY_DIR}")
target_link_libraries(lilith-host PRIVATE Qt6::Widgets Qt6::DBus LayerShellQt::Interface KF6::IdleTime X11::X11 X11::Xext)
add_executable(shm-layout tools/shm-layout.c)
add_custom_target(protocol-layout ALL
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tools/write-layout.py" "$<TARGET_FILE:shm-layout>" "${CMAKE_SOURCE_DIR}/build/shm-layout.json"
  DEPENDS shm-layout VERBATIM)
include(CTest)
if(BUILD_TESTING)
  find_program(DBUS_RUN_SESSION dbus-run-session REQUIRED)
  foreach(name motion-guard rebound-guard transport-stress)
    add_executable(${name} "tests/${name}.c")
    target_compile_options(${name} PRIVATE -UNDEBUG)
    target_link_libraries(${name} PRIVATE Threads::Threads)
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
  foreach(name global-shortcuts desktop-bridge tray-bridge native-bridge)
    add_executable(${name}-test "tests/${name}-test.cpp")
    add_dependencies(${name}-test bridge-moc)
    target_include_directories(${name}-test PRIVATE "${CMAKE_BINARY_DIR}")
    target_compile_options(${name}-test PRIVATE -UNDEBUG)
    target_link_libraries(${name}-test PRIVATE Qt6::Widgets Qt6::DBus KF6::IdleTime)
    add_test(NAME ${name} COMMAND "${DBUS_RUN_SESSION}" -- "$<TARGET_FILE:${name}-test>")
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 45)
  endforeach()
  add_test(NAME installer-safety COMMAND "${Python3_EXECUTABLE}" -m unittest discover -s tests -p "test_*.py")
  set_tests_properties(installer-safety PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 45)
endif()
